// pages/ai/ai.js
Page({
  data: {
    inputVal: ''
  },
  onInput: function(e) {
    this.setData({
      inputVal: e.detail.value
    });
  },
  onSend: function() {
    const inputVal = this.data.inputVal;
    wx.cloud.callFunction({
      name: 'generateText',
      data: { text: inputVal },
      success: res => {
        this.setData({
          inputVal: res.result.text
        });
      }
    });
  }
});
