Page({
  data: {
    question: '',
    response: ''
  },
  inputHandler: function(e) {
    this.setData({
      question: e.detail.value
    });
  },
  sendQuestion: function() {
    const question = this.data.question;
    if (question) {
      // 模拟请求AI模型处理问题
      this.setData({
        response: "正在思考..."
      });
      setTimeout(() => {
        this.setData({
          response: "这是AI的回答：你的问题是什么？"
        });
      }, 2000);
    }
  }
})
