```javascript
// pages/ai/ai.js
Page({
  data: {
    inputText: '',
    reply: ''
  },
  onInput: function(event) {
    this.setData({
      inputText: event.detail.value
    });
  },
  onSubmit: function() {
    wx.cloud.callFunction({
      name: 'aiChat',
      data: {
        text: this.data.inputText
      },
      success: res => {
        this.setData({
          reply: res.result.text
        });
      },
      fail: err => {
        console.error('Call AI function failed', err);
      }
    });
  }
});
```
