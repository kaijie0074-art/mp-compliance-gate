===FILE pages/ai/ai.js===
Page({
  data: {
    question: '',
    result: ''
  },
  onInput: function(e) {
    this.setData({
      question: e.detail.value
    });
  },
  submitQuestion: function() {
    // 这里应该调用AI接口，为了演示，我们使用本地数据
    const answers = {
      "What is AI?": "Artificial Intelligence",
      "What is ML?": "Machine Learning"
    };
    this.setData({
      result: answers[this.data.question] || "I don't know."
    });
  }
})
