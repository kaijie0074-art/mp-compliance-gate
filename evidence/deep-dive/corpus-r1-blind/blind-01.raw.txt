Page({
  data: {
    question: '',
    answer: '',
    showAnswer: false
  },
  onQuestionInput: function(event) {
    this.setData({
      question: event.detail.value
    });
  },
  submitQuestion: function() {
    // 此处应该调用 API 获取答案，但为了简单起见，这里仅作为占位符
    this.setData({
      answer: "这是根据您的提问生成的回答。",
      showAnswer: true
    });
  }
})
