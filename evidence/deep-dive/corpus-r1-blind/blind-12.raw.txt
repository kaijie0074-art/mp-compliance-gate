===FILE pages/ai/ai.js===
Page({
  data: {
    question: '',
    result: ''
  },
  onInput: function(e) {
    this.setData({
      question: e.detail.value
    });
  },
  onQuery: function() {
    wx.request({
      url: 'http://example.com/api/ai',
      method: 'GET',
      data: {
        question: this.data.question
      },
      success: (res) => {
        this.setData({
          result: res.data.answer
        });
      }
    });
  }
})
